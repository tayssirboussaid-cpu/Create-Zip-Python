import os
from flask import Flask, request, redirect, url_for, render_template_string, send_from_directory, flash, session
from flask_sqlalchemy import SQLAlchemy
from werkzeug.utils import secure_filename
from datetime import datetime

BASE_DIR = os.getcwd()
UPLOAD_FOLDER = os.path.join(BASE_DIR, 'uploads')
DB_PATH = os.path.join(BASE_DIR, 'reports.db')
ALLOWED_EXT = {'png', 'jpg', 'jpeg', 'pdf'}

os.makedirs(UPLOAD_FOLDER, exist_ok=True)

app = Flask(__name__)
app.config['SECRET_KEY'] = 'dev-secret-key'
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///{}'.format(DB_PATH)
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False
app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER

db = SQLAlchemy(app)

USERS = {'admin':'12345'}

@app.route('/login', methods=['GET','POST'])
def login():
    if request.method=='POST':
        username = request.form.get('username')
        password = request.form.get('password')
        if username in USERS and USERS[username]==password:
            session['user'] = username
            flash('تم تسجيل الدخول بنجاح')
            return redirect(url_for('index'))
        else:
            flash('اسم المستخدم أو كلمة السر غير صحيحة')
            return redirect(url_for('login'))
    return render_template_string(BASE_HTML + """
    {% block content %}
      <h2>تسجيل الدخول</h2>
      <form method="post">
        <label>اسم المستخدم:</label><input name="username"><br>
        <label>كلمة السر:</label><input type="password" name="password"><br>
        <button type="submit">دخول</button>
      </form>
    {% endblock %}
    """)

@app.before_request
def require_login():
    if request.endpoint not in ['login', 'uploaded_file', 'static'] and 'user' not in session:
        return redirect(url_for('login'))

# ----- Models -----
class Center(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    ministry = db.Column(db.String(200))
    administration = db.Column(db.String(200))
    center_name = db.Column(db.String(200))
    city = db.Column(db.String(100))
    state = db.Column(db.String(100))
    director = db.Column(db.String(200))
    exam_date = db.Column(db.String(50))
    logo_filename = db.Column(db.String(255))

class Stream(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    code = db.Column(db.String(50), unique=True)
    name = db.Column(db.String(200))

class Subject(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    stream_id = db.Column(db.Integer, db.ForeignKey('stream.id'))
    name = db.Column(db.String(200))
    stream = db.relationship('Stream', backref='subjects')

class Institute(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(200))

class Teacher(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(200))
    subject_id = db.Column(db.Integer, db.ForeignKey('subject.id'), nullable=True)
    max_reports = db.Column(db.Integer, default=100)
    subject = db.relationship('Subject', backref='teachers')

class Series(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    series_number = db.Column(db.String(50))
    reports_count = db.Column(db.Integer)
    subject_id = db.Column(db.Integer, db.ForeignKey('subject.id'))
    subject = db.relationship('Subject', backref='series')

class Report(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    candidate_name = db.Column(db.String(200))
    registration_no = db.Column(db.String(100))
    subject_id = db.Column(db.Integer, db.ForeignKey('subject.id'))
    file_name = db.Column(db.String(255))
    series_id = db.Column(db.Integer, db.ForeignKey('series.id'), nullable=True)
    subject = db.relationship('Subject')
    series = db.relationship('Series')

class Assignment(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    report_id = db.Column(db.Integer, db.ForeignKey('report.id'))
    teacher_id = db.Column(db.Integer, db.ForeignKey('teacher.id'))
    status = db.Column(db.String(50), default='assigned')
    grade = db.Column(db.Float, nullable=True)
    session = db.Column(db.String(50), default='main')
    report = db.relationship('Report')
    teacher = db.relationship('Teacher')

class Accounting(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    teacher_id = db.Column(db.Integer, db.ForeignKey('teacher.id'))
    reports_count = db.Column(db.Integer)
    price_per_report = db.Column(db.Float)
    amount = db.Column(db.Float)
    session = db.Column(db.String(50))
    teacher = db.relationship('Teacher')

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXT

# ---- Initialize DB ----
@app.before_request
def setup_defaults():
    db.create_all()
    if Stream.query.count() == 0:
        streams = [
            ('literature','الآداب'), ('mathematics','الرياضيات'), ('experimental_sciences','العلوم التجريبية'),
            ('economics','الاقتصاد والتصرف'), ('technical_sciences','العلوم التقنية'), ('informatics','علوم الإعلامية'), ('sports','الرياضة')
        ]
        for code, name in streams:
            db.session.add(Stream(code=code, name=name))
        db.session.commit()
